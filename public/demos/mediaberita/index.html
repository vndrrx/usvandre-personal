Arsitektur Database MySQL `mediaberita` (6 Tabel Relasional)
berita
  • id : int(11) PK
  • judul_berita : varchar(250)
  • isi_berita : text
  • tanggal_post : date
  • penulis_berita : int(11) FK
  • gambar : varchar(255)
  • views_count : int(11)
  • comment_count : int(11)
users
  • id : int(11) PK
  • username : varchar(20)
  • password : varchar(255)
  • gambar : varchar(255)
kategori_berita
  • id : int(11) PK
  • nama : varchar(255)
berita_kategori
  • berita_id : int(11) FK
  • kategori_id : int(11) FK
comment
  • id : int(11) PK
  • post_id : int(11) FK
  • name : varchar(255)
  • email : varchar(255)
  • message : text
  • created_at : timestamp
views
  • id : int(11) PK
  • post_id : int(11) FK
  • views_count : int(11)
